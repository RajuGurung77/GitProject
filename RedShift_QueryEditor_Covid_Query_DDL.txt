create schema if not exists covid_gold;

CREATE TABLE covid_gold.dim_date (
  date_id int,
  full_date timestamp,
  year int,
  month int,
  day int,
  dow int,
  is_weekend boolean
);

CREATE TABLE IF NOT EXISTS covid_gold.dim_state (
  state_code varchar(2) PRIMARY KEY,
  state_name varchar(64)
)
DISTSTYLE ALL;

COPY covid_gold.dim_date
FROM 's3://covid19-takeo/covid/gold/dim_date/'
IAM_ROLE 'arn:aws:iam::023703349379:role/RedShift_Admin'
FORMAT AS PARQUET;



COPY covid_gold.dim_state
FROM 's3://covid19-takeo/covid/gold/dim_state/'
IAM_ROLE 'arn:aws:iam::023703349379:role/RedShift_Admin'
FORMAT AS PARQUET;


CREATE TABLE covid_gold.fact_cases_state_daily (
  date_id int,
  state_code varchar(2),
  cases_cum bigint,
  deaths_cum bigint,
  new_cases bigint,
  new_deaths bigint
)
DISTKEY(state_code)
SORTKEY(date_id, state_code);

COPY covid_gold.fact_cases_state_daily
FROM 's3://covid19-takeo/covid/gold/fact_cases_state_daily/'
IAM_ROLE 'arn:aws:iam::023703349379:role/RedShift_Admin'
FORMAT AS PARQUET;

CREATE TABLE covid_gold.fact_testing_state_daily (
  date_id int,
  state_code varchar(2),
  tests_total_cum bigint,
  tests_pos_cum bigint,
  tests_neg_cum bigint,
  new_tests bigint,
  positivity_rate double precision
)
DISTKEY(state_code)
SORTKEY(date_id, state_code);

COPY covid_gold.fact_testing_state_daily
FROM 's3://covid19-takeo/covid/gold/fact_testing_state_daily/'
IAM_ROLE 'arn:aws:iam::023703349379:role/RedShift_Admin'
FORMAT AS PARQUET;
